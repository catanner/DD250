#Drop columns with identical values to reduce clutter in final output
def drop_identical_columns(df: pd.DataFrame) -> pd.DataFrame:
    cols = list(df.columns)
    to_drop = set()

    for i, c1 in enumerate(cols):
        if c1 in to_drop:
            continue
        for c2 in cols[i + 1:]:
            if c2 in to_drop:
                continue
            # True when values are identical (NaN in same positions also counts as equal)
            if df[c1].equals(df[c2]):
                to_drop.add(c2)

    return df.drop(columns=list(to_drop))
